* Move current line up or down
  #+BEGIN_SRC emacs-lisp
    (defun move-line-up ()
      "Move up the current line."
      (interactive)
      (transpose-lines 1)
      (forward-line -2)
      (indent-according-to-mode))

    (defun move-line-down ()
      "Move down the current line."
      (interactive)
      (forward-line 1)
      (transpose-lines 1)
      (forward-line -1)
      (indent-according-to-mode))

    (global-set-key [(control shift up)]  'move-line-up)
    (global-set-key [(control shift down)]  'move-line-down)
  #+END_SRC

* Hide undo-tree history files from find-file completion
#+BEGIN_SRC emacs-lisp
  ;; ivy lists files with `all-completions', which skips
  ;; `completion-ignored-extensions', so filter its list directly.
  (define-advice ivy--sorted-files (:filter-return (files) hide-undo-tree)
    (cl-remove-if (lambda (f) (string-suffix-p ".~undo-tree~" f)) files))
#+END_SRC
* Keep undo-tree files in one directory and clean old ones
  Save histories in =~/.emacs.d/undo= instead of next to each file, and
  delete those older than a week, once, after Emacs has been idle for a
  few seconds. Listing one directory is cheap.
#+BEGIN_SRC emacs-lisp
  (defvar undo-history-directory (expand-file-name "undo" user-emacs-directory))
  (make-directory undo-history-directory t)
  (setq undo-tree-history-directory-alist `(("." . ,undo-history-directory)))

  (defun clean-undo-tree-files (&optional days)
    "Delete undo-tree files in `undo-history-directory' older than DAYS (default 7)."
    (let ((cutoff (time-subtract nil (days-to-time (or days 7)))))
      (dolist (f (directory-files undo-history-directory t "\\.~undo-tree~\\'"))
        (when (time-less-p (file-attribute-modification-time (file-attributes f))
                           cutoff)
          (delete-file f)))))

  (run-with-idle-timer 10 nil #'clean-undo-tree-files)
#+END_SRC
